from datetime import datetime
from flask_login import UserMixin
from apps.app import db
 
 
class Recruitment(db.Model):
    """募集テーブル"""
    __tablename__ = "recruitments"
 
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, db.ForeignKey("users.id"), nullable=False)
    title = db.Column(db.String, nullable=False)
    description = db.Column(db.String, nullable=False)
    photo = db.Column(db.LargeBinary, nullable=False)
    location_prefecture = db.Column(db.String, nullable=False)
    location_city = db.Column(db.String, nullable=False)
    location_detail = db.Column(db.String, nullable=False)
    start_at = db.Column(db.DateTime, nullable=False)
    end_at = db.Column(db.DateTime, nullable=False)
    application_start_at = db.Column(db.DateTime, nullable=False)
    application_end_at = db.Column(db.DateTime, nullable=False)
    capacity = db.Column(db.Integer, nullable=False)
    eligibility = db.Column(db.String, nullable=False)
    required_items = db.Column(db.String, nullable=False)
    status = db.Column(db.String, nullable=False)
    created_at = db.Column(db.DateTime, nullable=False, default=datetime.now)
    updated_at = db.Column(db.DateTime, default=datetime.now, onupdate=datetime.now)
 
    user = db.relationship("User", back_populates="recruitments")
    likes = db.relationship("Like", back_populates="recruitment", cascade="all, delete-orphan")
    view_histories = db.relationship("ViewHistory", back_populates="recruitment", cascade="all, delete-orphan")
    tags = db.relationship("Tag", back_populates="recruitment", cascade="all, delete-orphan")
    search_histories = db.relationship("SearchHistory", back_populates="recruitment", cascade="all, delete-orphan")
    participation_histories = db.relationship("ParticipationHistory", back_populates="recruitment", cascade="all, delete-orphan")
    recruitment_histories = db.relationship("RecruitmentHistory", back_populates="recruitment", cascade="all, delete-orphan")
    active_messages = db.relationship("ActiveMessage", back_populates="recruitment", cascade="all, delete-orphan")

@dt.route("/for_active/<recruitment_id>", methods=["GET", "POST"])
@login_required
def for_active(recruitment_id):
    # 募集情報を取得
    recruitment = get_recruitment(recruitment_id)

    # メッセージ投稿フォーム
    form = MessageForm()

    # POSTリクエスト時の処理分岐
    if request.method == "POST":
        #メッセージ投稿ボタンが押された場合
        if form.submit.name in request.form:
            if form.validate_on_submit():
                message = ActiveMessage(
                    recruitment_id=recruitment_id, 
                    user_id=get_user_id(),
                    message=form.message.data,
                )
                db.session.add(message)
                db.session.commit()
                flash("メッセージを投稿しました")
                return redirect(url_for("dt.for_active", recruitment_id=recruitment_id))

        #参加者削除ボタンが押された場合
        elif "delete_participation" in request.form:
            participation_id = request.form.get("participation_id")
            entry = ParticipationHistory.query.filter_by(
                id=participation_id
            ).first_or_404()

            #募集者本人かチェック
            if recruitment.user_id != get_user_id():
                abort(403)

            db.session.delete(entry)
            db.session.commit()
            flash("申込者を削除しました")
            return redirect(url_for("dt.for_active", recruitment_id=recruitment_id))

    # 参加者一覧を取得
    entries = ParticipationHistory.query.filter_by(
        recruitment_id=recruitment_id
    ).all()

    return render_template(
        "natyukatu/for_active.html",
        recruitment=recruitment,
        entries=entries,
        form=form,
    )

def delete_participation(participation_id):

    ).first_or_404()
 
    recruitment = get_recruitment(entry.recruitment_id)
 
    if recruitment.user_id != get_user_id():
        abort(403)
